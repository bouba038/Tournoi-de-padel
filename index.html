function selectWinner(section, round, matchId, teamNum) { if (!state.isOrgSession) { alert("Mode lecteur : vous ne pouvez pas modifier les vainqueurs."); return; } let matchArray; if (section === "prelim") matchArray = state.bracket.prelim; else matchArray = state.bracket[section][round]; const matchIndex = matchArray.findIndex(m => m.id === matchId); if (matchIndex === -1) return; const match = matchArray[matchIndex]; if (!match.t1 || !match.t2) return; const winner = teamNum === 1 ? match.t1 : match.t2; const loser = teamNum === 1 ? match.t2 : match.t1; match.winner = winner; if (state.bracket.size === 16) { if (section === "prelim") { // Propagation vers R1 state.bracket.principal.R1[matchIndex].t2 = winner; // Consolante 13-16 (Paires: 0&1 -> SF 0 ; 2&3 -> SF 1) const sf13Idx = matchIndex < 2 ? 0 : 1; const isSlot2 = matchIndex % 2 !== 0; const sf13Match = state.bracket.consolante.SF13_16[sf13Idx]; if (!isSlot2) sf13Match.t1 = loser; else sf13Match.t2 = loser; } else if (section === "principal" && round === "R1") { // Propagation vers QF state.bracket.principal.QF[matchIndex].t2 = winner; // Consolante 9-12 const consIdx = matchIndex < 2 ? 0 : 1; const isSlot2 = matchIndex % 2 !== 0; const consMatch = state.bracket.consolante.SF[consIdx]; if (!isSlot2) consMatch.t1 = loser; else consMatch.t2 = loser; } else if (section === "principal" && round === "QF") { // Vers Demies 5-8 const p58Idx = matchIndex < 2 ? 0 : 1; const isSlot2 = matchIndex % 2 !== 0; const p58Match = state.bracket.places5_8.SF[p58Idx]; if (!isSlot2) p58Match.t1 = loser; else p58Match.t2 = loser; // Vers Demies Principales (0,1 -> SF 0 ; 2,3 -> SF 1) const sfIdx = matchIndex < 2 ? 0 : 1; const sfSlot2 = matchIndex % 2 !== 0; const sfMatch = state.bracket.principal.SF[sfIdx]; if (!sfSlot2) sfMatch.t1 = winner; else sfMatch.t2 = winner; } else if (section === "principal" && round === "SF") { const fMatch = state.bracket.principal.F[0]; if (matchIndex === 0) fMatch.t1 = winner; else fMatch.t2 = winner; const thirdMatch = state.bracket.principal.Third[0]; if (matchIndex === 0) thirdMatch.t1 = loser; else thirdMatch.t2 = loser; } else if (section === "places5_8" && round === "SF") { const f58 = state.bracket.places5_8.F[0]; if (matchIndex === 0) f58.t1 = winner; else f58.t2 = winner; const s7 = state.bracket.places5_8.Seventh[0]; if (matchIndex === 0) s7.t1 = loser; else s7.t1 = loser; // Corrigé ici } else if (section === "consolante") { if (round === "SF13_16") { const t13 = state.bracket.consolante.Thirteenth[0]; if (matchIndex === 0) t13.t1 = winner; else t13.t2 = winner; const t15 = state.bracket.consolante.Fifteenth[0]; if (matchIndex === 0) t15.t1 = loser; else t15.t2 = loser; } else if (round === "SF") { const fCons = state.bracket.consolante.F[0]; if (matchIndex === 0) fCons.t1 = winner; else fCons.t2 = winner; const t11 = state.bracket.consolante.Eleventh[0]; if (matchIndex === 0) t11.t1 = loser; else t11.t2 = loser; } } } render(); }
